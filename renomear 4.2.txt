local function loadConfig()
    if isfile and isfile(FILE) then
        local ok, data = pcall(function() return HttpService:JSONDecode(readfile(FILE)) end)
        if ok and type(data) == "table" then
            for k, v in pairs(data) do if Default[k] ~= nil then Config[k] = v end end
        end
    end
end
local function saveConfig()
    if writefile then
        pcall(function() writefile(FILE, HttpService:JSONEncode(Config)) end)
    end
end

local function writeAutoExec()
    if not (writefile and makefolder) then return false end
    local folders = { "autoexec", "auto_execute", "autoexecute" }
    local body = ([[
-- GHOST HUB AUTO CHEST (auto execute)
task.wait(6)
local cfg = [==[%s]==]
pcall(function() writefile("%s", cfg) end)
local ok, src = pcall(function() return game:HttpGet("%s") end)
if ok and src and #src > 200 then
    pcall(function() loadstring(src)() end)
end
]]):format(HttpService:JSONEncode(Config), FILE, SCRIPT_URL)
    local wrote = false
    for _, f in ipairs(folders) do
        pcall(function()
            if not isfolder or not isfolder(f) then makefolder(f) end
            writefile(f .. "/GhostHub_AutoChest.lua", body)
            wrote = true
        end)
    end
    return wrote
end

local langFileLoaded = false
local function loadLangFile()
    if not (isfile and isfile(LANG_FILE)) then return end
    local ok, data = pcall(function() return HttpService:JSONDecode(readfile(LANG_FILE)) end)
    if ok and type(data) == "table" then
        for lang, tbl in pairs(data) do
            if type(tbl) == "table" then
                L[lang] = L[lang] or {}
                for k, v in pairs(tbl) do
                    if type(v) == "string" then L[lang][k] = v end
                end
            end
        end
        langFileLoaded = true
    end
end
local function exportLangFile()
    if not writefile then return false end
    local ok = pcall(function() writefile(LANG_FILE, HttpService:JSONEncode(L)) end)
    return ok
end

local FONT_LOGO = Enum.Font.Fondamento      -- letra com curvas (GHOST HUB)
-- ==== FONTE GLOBAL: Varela Round (fallback GothamMedium) ====
local UI_FONTFACE = nil
local UI_FONT_ENUM = Enum.Font.GothamMedium
do
    local ok, face = pcall(function()
        return Font.fromName("VarelaRound", Enum.FontWeight.Medium, Enum.FontStyle.Normal)
    end)
    if ok and face then UI_FONTFACE = face end
end
local function applyFont(obj)
    if UI_FONTFACE then
        local ok = pcall(function() obj.FontFace = UI_FONTFACE end)
        if ok then return end
    end
    pcall(function() obj.Font = UI_FONT_ENUM end)
end
-- aplica a fonte em tudo (menos a logo) e mantem aplicando no que nascer depois
local function applyFontTree(root)
    for _, d in ipairs(root:GetDescendants()) do
        if d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox") then
            if d.Font ~= FONT_LOGO then applyFont(d) end
        end
    end
    root.DescendantAdded:Connect(function(d)
        if d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox") then
            task.defer(function()
                if d.Font ~= FONT_LOGO then applyFont(d) end
            end)
        end
    end)
end
local function navyGradient(o, rot)
    local g = Instance.new("UIGradient")
    g.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, Color3.fromRGB(10, 18, 46)),
        ColorSequenceKeypoint.new(0.5, Color3.fromRGB(14, 26, 66)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(8, 14, 36)),
    })
    g.Rotation = rot or 90
    g.Parent = o
    return g
end

        corner(val, 5); stroke(val, C.cyan)

            local cur, idx = Config[field], 1
            for i, p in ipairs(presets) do
                if math.abs(p - cur) < 1e-6 then idx = i break end
                if p > cur then idx = i - 1 break end
                idx = i
            end
            local nxt = presets[(idx % #presets) + 1]
            Config[field] = math.clamp(nxt, min, max)
            if paintRef then paintRef() end
            saveConfig()
            if onChange then task.spawn(onChange, Config[field]) end
        end)
    end

    local bar = Instance.new("Frame")

        local v = Config[field]
        val.Text = fmt(v)
        if not box:IsFocused() then box.Text = fmt(v) end
        fill.Size = UDim2.new(math.clamp((v - min) / (max - min), 0, 1), 0, 1, 0)
    end

    box:GetPropertyChangedSignal("Text"):Connect(function()
        local n = tonumber((box.Text or ""):gsub(",", "."))
        if n then
            n = math.clamp(n, min, max)
            val.Text = fmt(n)
            fill.Size = UDim2.new(math.clamp((n - min) / (max - min), 0, 1), 0, 1, 0)
        end
    end)

    local function setFromX(px)
        local a = math.clamp((px - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        local v = min + (max - min) * a
        if decimals and decimals > 0 then
            local m = 10 ^ decimals; v = math.floor(v * m + 0.5) / m
        else v = math.floor(v + 0.5) end

            if dragging then

            end
        end
    end)

        local n = tonumber((box.Text or ""):gsub(",", "."))

            Config[field] = math.clamp(n, min, max)
            saveConfig()
            if onChange then task.spawn(onChange, Config[field]) end
        end

    end)

        elseif mode == "Box" then
            f.Size = UDim2.new(1, 0, 0, 52)

        else -- TargetBox
            f.Size = UDim2.new(1, 0, 0, 52)
            bar.Visible = false
            val.Visible = true; box.Visible = true
        end

local simpleCache = {}
local function simplifyChest(model, kind)
    if simpleCache[model] then return end
    local base = rootOf(model); if not base then return end
    local cf, size = base.CFrame, base.Size
    if model:IsA("Model") then
        local ok, c, s = pcall(function() local a, b = model:GetBoundingBox(); return a, b end)
        if ok and c then cf, size = c, s end
    end
    local hidden = {}
    for _, d in ipairs(model:GetDescendants()) do
        if d:IsA("BasePart") then
            table.insert(hidden, { obj = d, t = d.Transparency })
            d.Transparency = 1
        elseif d:IsA("SpecialMesh") or d:IsA("Decal") or d:IsA("Texture")
            or d:IsA("ParticleEmitter") or d:IsA("Beam") or d:IsA("Trail") then
            table.insert(hidden, { obj = d, enabled = true })
            pcall(function() d.Enabled = false end)
            pcall(function() d.Transparency = 1 end)
        end
    end
    local box = Instance.new("Part")
    box.Name = "GhostHubSimple"
    box.Anchored = true
    box.CanCollide = false
    box.Material = Enum.Material.Neon
    box.Color = SIMPLE_COLORS[kind] or C.cyan
    box.Size = size
    box.CFrame = cf
    box.Parent = model
    simpleCache[model] = { box = box, hidden = hidden }
end
local function restoreChest(model)
    local data = simpleCache[model]; if not data then return end
    pcall(function() data.box:Destroy() end)
    for _, h in ipairs(data.hidden) do
        pcall(function()
            if h.t ~= nil then h.obj.Transparency = h.t end
            if h.enabled then h.obj.Enabled = true end
        end)
    end
    simpleCache[model] = nil
end
local function restoreAllChests()
    for m in pairs(simpleCache) do restoreChest(m) end
end

-- lista de servers PUBLICOS, ja filtrando privado/VIP, cheio e limites de jogadores
local function fetchServers()
    local found, cursor, pages = {}, nil, 0
    repeat
        pages = pages + 1
        local url = "https://games.roblox.com/v1/games/" .. game.PlaceId ..
            "/servers/Public?sortOrder=Asc&limit=100" .. (cursor and ("&cursor=" .. cursor) or "")
        local ok, body = pcall(function() return game:HttpGet(url) end)
        if not ok then break end
        local okd, data = pcall(function() return HttpService:JSONDecode(body) end)
        if not okd or type(data) ~= "table" or not data.data then break end
        for _, s in ipairs(data.data) do
            local playing  = tonumber(s.playing) or 0
            local maxp     = tonumber(s.maxPlayers) or 0
            -- server privado / VIP: tem accessCode ou vipServerId, ou nao aparece como publico
            local private  = (s.accessCode ~= nil) or (s.vipServerId ~= nil) or (s.private == true)
            local okPriv   = not (Config.AvoidPrivate and private)
            local okRoom   = maxp > 0 and playing < maxp
            local okMax    = playing <= math.max(1, Config.HopMaxPlayers or 20)
            local okMin    = playing >= math.max(0, Config.HopMinPlayers or 0)
            if s.id ~= game.JobId and okPriv and okRoom and okMax and okMin then
                table.insert(found, { id = s.id, playing = playing, ping = tonumber(s.ping) or 9999 })
            end
        end
        cursor = data.nextPageCursor
        task.wait(0.15)
    until (not cursor) or pages >= 5 or #found >= 120
    return found
end

local function pickServer(list)
    if #list == 0 then return nil end
    local mode = Config.HopSelect or "latency"
    if mode == "latency" then
        table.sort(list, function(a, b) return a.ping < b.ping end)
        return list[math.random(1, math.min(5, #list))].id
    elseif mode == "players" then
        table.sort(list, function(a, b) return a.playing < b.playing end)
        return list[math.random(1, math.min(5, #list))].id
    end
    return list[math.random(#list)].id
end

local function hopServer()
    hopQueue()
    local list = fetchServers()
    local id = pickServer(list)
    if id then
        local ok = pcall(function()
            TeleportService:TeleportToPlaceInstance(game.PlaceId, id, LP)
        end)
        if ok then return true end
    end
    -- nada compativel: nao cai em server privado, tenta de novo depois
    return false
end

-- pede hop mostrando a tela e permitindo cancelar
local function requestHop(reason)
    if hopPending then return end
    hopPending, hopCancel = true, false
    if Config.HopOverlay then
        local secs = math.max(1, math.floor(Config.HopDelay or 3))
        hopTitle.Text = T("hopping") .. (reason and ("  ·  " .. reason) or "")
        hopHint.Text = T("hopCancel")
        hopScreen.Visible = true
        for i = secs, 1, -1 do
            if hopCancel then break end
            hopSub.Text = ("%s  %d s"):format(T("hopIn"), i)
            local t0 = os.clock()
            while os.clock() - t0 < 1 do
                if hopCancel then break end
                task.wait(0.05)
            end
        end
        hopScreen.Visible = false
        if hopCancel then
            hopPending = false
            return false
        end
    end
    local done = hopServer()
    if not done then
        hopTitle.Text = T("hopNone")
        hopSub.Text = ""
        hopHint.Text = ""
        hopScreen.Visible = true
        task.wait(2)
        hopScreen.Visible = false
    end
    hopPending = false
    return done
end
